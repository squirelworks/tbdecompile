to handle enterSystem cmdLine
	system int ntries
	system string firstResponse
		
	ntries = 0
	firstResponse = ""
	
	if "tb80r.sbk" is not in sysBooks
		push "tb80r.sbk" onto sysBooks
	end

	if "nc81base.sbk" is not in sysBooks
		push "nc81base.sbk" onto sysBooks
		get initializebase(cmdLine)
	end
	if "nc81test.sbk" is not in sysBooks
		push "nc81test.sbk" onto sysBooks
		get initializetest()
	end
	if "nc81recs.sbk" is not in sysBooks
		push "nc81recs.sbk" onto sysBooks
		get initializerecs()	
	end	
	if "nc81trnr.sbk" is not in sysBooks
		push "nc81trnr.sbk" onto sysBooks
		get initializetrnr()	
	end	
	
	get openSession()	
	if it is 1  then
		get gotoPageList()
	end if
	
	forward
end

to handle keyup key
	system int __quizHelp
	system int __quizDisableMenuKey
	
	if  __quizHelp = 0  then
		conditions 
			when key = keyf1
		    	request "Do you want to go to the Key Map?" with "&Yes" or "&No"
				if it is "Yes" then
					open viewer "key map"
					currentpage of viewer "key map" = page "P20000" of book "nida81.tbk" 
					show viewer "key map"
					break
					send next
				end if
			when key = keyf2
				request "Are you sure you want to Exit the Lesson?" with "&Yes" or "&No"
				if it is "Yes" then
					get EndExperiment()
			       	get closeSession()
    	    	send exit  
	      		end if
			when key = keyf4
				if  __quizDisableMenuKey = 0  then
					request "Do you want to go to the Lesson Menu?" with "&Yes" or "&No"
			   		if it is "Yes" then
						get EndExperiment()
						get gotoMenuPage()
					end if
				end if
			when key = keyf5
				get specialFunctions(key)
			when key = keyf6
				get specialFunctions(key)	
			when key = keyf7
				get specialFunctions(key)
			when key = keyf8
				get specialFunctions(key)
		    when key = keyf9
		    	if getExperimentStatus() = 1  then
		    		beep(1)
		    	else
					request "Are you sure you want to Bookmark the Lesson?" with "&Yes" or "&No"
					if it is "Yes" then
						get EndExperiment()
						get bookmarkSession()
						send exit  
					end if
				end if
			--when key = keyf10
				--get specialFunctions(key)
			when key = keyf11
				get specialFunctions(key)
			when key = keyf12
				get specialFunctions(key)
			when key = keyEnter
      			if  isObject(button("Next"))  then
			      	send buttonClick to button "next"
				end if
	    else
			forward
		end conditions
	else
		conditions
			when key = keyEnter
      			if  isObject(button("Next"))  then
			      	send buttonClick to button "next"
				end if
			when key = keyF5
				get returnToQuiz()				
		else
			forward
		end conditions
	end if
end
