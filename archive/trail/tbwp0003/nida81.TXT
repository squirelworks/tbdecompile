to handle keyup key
system lastbook
  conditions 
	when key = keyf1
    	request "Do you want to go to the Key Map?" with "&Yes" or "&No"
			if it is "Yes" then
					open viewer "key map"
					currentpage of viewer "key map" = page "P20000" of book "nida81.tbk" 
					show viewer "key map"
				break
			send next
			end if
    when key = keyf2
		request "Are you sure you want to Exit the Lesson?" with "&Yes" or "&No"
			if it is "Yes" then
				get EndExperiment()
		       	get closeSession()
    	    send exit  
      		end if
 	when key = keyf4
    	request "Do you want to go to the Lesson Menu?" with "&Yes" or "&No"
    		if it is "Yes" then
				get EndExperiment()
				go to page "p900" of book lastbook
      end if

	when key = keyf5
		get specialFunctions(key)

	when key = keyf6
		get specialFunctions(key)	

	when key = keyf7
		get specialFunctions(key)

	when key = keyf8
		get specialFunctions(key)

    when key = keyf9
      request "Are you sure you want to Bookmark the Lesson?" with "&Yes" or "&No"
      if it is "Yes" then
        get EndExperiment()
      	get bookmarkSession()
        send exit  
      end if

	--when key = keyf10
	--	get specialFunctions(key)

	when key = keyf11
		get specialFunctions(key)

when key = keyf12
		get specialFunctions(key)
	when key = keyEnter
      	if  isObject(button("Next"))  then
      		if  visible of button("Next") is true  and  enabled of button("Next") is true  then
	      		send buttonClick to button "next"
	      	end if
      	end if
    else
      forward
  end conditions
end

To handle enterpage
	system trainersettings
	system ntries
	
	ntries = 0
	
	set text of field cnlv to item 1 of trainersettings
	set text of field cplv to item 2 of trainersettings
	
	set text of field cac1 to "OFF"
	set text of field cdc1 to "OFF"
	set text of field cac2 to "OFF"
	set text of field cdc2 to "OFF"
	set text of field cspk to "OFF"
	set text of field cdc3 to "OFF"
	set text of field cfil to "OFF"
	
	trainersettings = uppercase(trainersettings)
	step i from 1 to charCount(trainersettings)
		if char i of trainersettings is ","  then
			set char i of trainersettings to " "
		end if
	end step
	
	step i from 3 to wordCount(trainersettings)
		temp = word i of trainersettings
		conditions
			when temp is "AC1"
				set text of field cac1 to "ON"		
			when temp is "DC1"
				set text of field cdc1 to "ON"		
			when temp is "AC2"
				set text of field cac2 to "ON"		
			when temp is "DC2"
				set text of field cdc2 to "ON"		
			when temp is "SPK"
				set text of field cspk to "ON"		
			when temp is "DC3"
				set text of field cdc3 to "ON"		
			when temp is "FIL"
				set text of field cfil to "ON"		
		end conditions
	end step
	
	send updatesettings to self
	forward
end
