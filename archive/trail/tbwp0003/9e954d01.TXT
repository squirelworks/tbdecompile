to handle keyup key
	system int __quizHelp
	system int __quizDisableMenuKey
	
	if  __quizHelp = 0  then
		conditions 
			when key = keyf1
		    	request "Do you want to go to the Key Map?" with "&Yes" or "&No"
				if it is "Yes" then
					open viewer "key map"
					currentpage of viewer "key map" = page "P20000" of book "nida81.tbk" 
					show viewer "key map"
					break
					send next
				end if
			when key = keyf2
				request "Are you sure you want to Exit the Lesson?" with "&Yes" or "&No"
				if it is "Yes" then
					get EndExperiment()
			       	get closeSession()
    	    	send exit  
	      		end if
			when key = keyf4
				if  __quizDisableMenuKey = 0  then
					request "Do you want to go to the Lesson Menu?" with "&Yes" or "&No"
			   		if it is "Yes" then
						get EndExperiment()
						get gotoMenuPage()
					end if
				end if
			when key = keyf5
				get specialFunctions(key)
			when key = keyf6
				get specialFunctions(key)	
			when key = keyf7
				get specialFunctions(key)
			when key = keyf8
				get specialFunctions(key)
		    when key = keyf9
		    	if getExperimentStatus() = 1  then
		    		beep(1)
		    	else
					request "Are you sure you want to Bookmark the Lesson?" with "&Yes" or "&No"
					if it is "Yes" then
						get EndExperiment()
						get bookmarkSession()
						send exit  
					end if
				end if
			--when key = keyf10
				--get specialFunctions(key)
			when key = keyf11
				get specialFunctions(key)
			when key = keyf12
				get specialFunctions(key)
			when key = keyEnter
      			if  isObject(button("Next"))  then
			      	send buttonClick to button "next"
				end if
	    else
			forward
		end conditions
	else
		conditions
			when key = keyEnter
      			if  isObject(button("Next"))  then
			      	send buttonClick to button "next"
				end if
			when key = keyF5
				get returnToQuiz()				
		else
			forward
		end conditions
	end if
end

sharedscript id 102 ("Question Accept")
----------------------------------------------------------------------
to handle buttonClick
	local int i
	local int error
	local string temp
	local int review
	
	local int score
	local string response
	
	response = text of field "response"
	
	if  response = ""  then
		beep 1
		focus = field "response"
		break
	end if
	
	if  response = answer of this page  then
		score = 100
	else
		score = 0
	end if

	error = storeQuestionResponse(testNo of this page, questionNo of this page, response, score)
	  
	review = getQuestionReview()
	i = getNextQuestion()
	if  i = -1  then
		if  review = 0  then
			temp = "P10300"
		else
	    	temp = "P10300"
	    end if
	else
		temp = "P" & (10100 + i)
	end if
  
	go to page temp
end

sharedscript id 101 ("Question Response")
----------------------------------------------------------------------
to handle keyChar key
	local string r
	local vObject
	
	if key = keyEnter  then
		break
	end if
	
	if  getQuizMode() = 0  then	
		r = uppercase(ansiToChar(key))
		conditions
			when r is in "A,B,C,D,E,F"
				vObject = r
				if  isObject(field(vObject))  then
					text of target = r
					focus = target
					caretLocation = 1,1
					break
				end if
		end conditions
	end if
	
	beep 1
end

sharedscript id 100 ("Question Page")
----------------------------------------------------------------------
to handle EnterPage
	local int review
	local int i
	local vObject
	local string temp
	system logical __import
	

	if  (sysLevel = "reader")  and  (__import = false)  then	
		focus = field "response"
		review = getQuestionReview()
		conditions
			when review = 0
				set visible of button "JumpBack" to false
				set visible of button "Accept" to false
				set visible of button "Back" to false
				set visible of button "JumpForward" to false
				set visible of button "Forward" to false
			when review = 1
				set visible of button "JumpBack" to false
				set visible of button "Back" to false
				set visible of button "JumpForward" to true
			when review = 2
				set visible of button "JumpBack" to true
				set visible of button "Back" to true
				set visible of button "JumpForward" to true
		end conditions
		set text of field "Response" to getQuestionResponse()
		set text of field "Index" to getQuestionIndex() & "."	
	else
		set visible of button "JumpBack" to true
		set visible of button "Accept" to true
		set visible of button "Back" to true
		set visible of button "JumpForward" to true
		set visible of button "Forward" to true
		set text of field "Response" to "?"
		set text of field "Index" to "999."	
	end if
	
	if getQuizMode() = 0  then
	--set visible of button "Accept" to true
		step i from 1 to 5
			vObject = AnsiToChar(CharToAnsi("A") + i - 1)
			if isObject(field(vObject))  then
				set strokeColor of field(vObject) to white
			end if	
		end step
	else
		if  sysLevel = "author"  then
			set visible of button "Accept" to true
		else
			set visible of button "Accept" to false
		end if
		step i from 1 to 5
			vObject = AnsiToChar(CharToAnsi("A") + i - 1)
			if isObject(field(vObject))  then
				temp = AnsiToChar(CharToAnsi("A") + i - 1)
				if  answer of this page = temp  then
					set strokeColor of field(vObject) to green
				else
					set strokeColor of field(vObject) to white
				end if
			end if	
		end step
	end if
	set text of field "page#" of this background to name of this page
end


to handle leavePage
	step i from 1 to 5
		vObject = AnsiToChar(CharToAnsi("A") + i - 1)
		if isObject(field(vObject))  then
			set strokeColor of field(vObject) to white
		end if	
	end step
	
	set text of field "response" to ""
end

to handle buttonClick
	local int i
	local vObject

	if getQuizMode() = 0  then	
		step i from 1 to 5
			vObject = AnsiToChar(CharToAnsi("A") + i - 1)
			if isObject(field(vObject))  then
				if target = field(vObject)  then
					send keyChar(CharToAnsi("A") + i - 1) to field "Response"
					break
				end if
			end if	
		end step
	end if
end

to handle keyUp key
	if  getQuizMode() = 0  then
		conditions
			when key is keyEnter
				send buttonClick to button "Accept"
		end conditions
	end if
end

----------------------------------------------------------------------
Page id 1017 ("P10301") of Book "C:\TBK\9e954d01.tbk"
----------------------------------------------------------------------
to handle enterPage
	if not isObject(button "A")  then
		forward
		break 
	end if

	set checked of button "A" to false
	set checked of button "B" to false
	
	if  sysLevel = author  then
		set text of field "stem" to "You skipped X question(s).  Do you want to _________"
		set enabled of button "A" to true
		set enabled of field "A" to true
	else
		skipped = getSkippedCount()
		conditions
			when  skipped = 0
				set text of field "stem" to "You skipped" && skipped && "questions.  Do you want to _________"
				set enabled of button "A" to false
				set enabled of field "A" to false
				set checked of button "B" to true
			when  skipped = 1
				set text of field "stem" to "You skipped" && skipped && "question.  Do you want to _________"
				set enabled of button "A" to true
